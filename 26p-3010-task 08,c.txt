#include <stdio.h>
int main() {
   double principal = 250000.00;  
    float rate = 8.5;             
    int time = 3;                 
   double interest = (principal * rate * time) / 100;
    double totalAmount = principal + interest;
   float monthlyInstallment = totalAmount / (time * 12.0);
 printf("============ BANK LOAN INTEREST SUMMARY =========\n");
  printf("Principal Amount:     PKR %.2f\n", principal);
 printf("Annual Interest Rate: %.2f%%\n", rate);
 printf("Loan Duration: %d Years (36 Months)\n", time);
 printf("-------------------------------------------------\n");
 printf("Total Accrued Interest: PKR %.2f\n", interest);
printf("Total Payable Amount:   PKR %.2f\n", totalAmount);
printf("Monthly Installment:  PKR %.2f\n", monthlyInstallment);
    printf("=================================================\n");
return 0;
}
